Clerk token facts (decision basis):
- Clerk session token `iss` is the Frontend API URL with NO trailing slash, e.g.
  `https://xxx.clerk.accounts.dev` (dev) or `https://clerk.<domain>` (prod). That is the
  same value CLERK_JWT_ISSUER_DOMAIN holds (it is already used to build the JWKS URL), so
  `issuer=domain.rstrip("/")` is correct.
- Clerk session tokens have no `aud` by default → keep `verify_aud: False`.
- `azp` (authorized party) is the requesting origin (e.g. `https://dispatch.example.com`)
  and may be absent (e.g. non-browser tokens) → check only when present.
- Dev sentinel path never reaches decode → unaffected by any of this.

    ```python
    _JWT_LEEWAY_S = 30

    def dashboard_allowed_origins() -> list[str]:
        """Parse DASHBOARD_ALLOWED_ORIGINS (comma-separated). [] when unset/blank. No default."""
        return [o.strip().rstrip("/") for o in os.environ.get("DASHBOARD_ALLOWED_ORIGINS", "").split(",") if o.strip()]

    def dev_mode_claims() -> dict | None:
        """None when CLERK_JWT_ISSUER_DOMAIN is set (verify for real).
        Unset + deployed -> HTTP 500 (same detail string as today). Unset + local -> warn, return sentinel."""

    async def verify_clerk_token(token: str) -> dict:
        """The ONLY Clerk jwt.decode site. Assumes CLERK_JWT_ISSUER_DOMAIN is set."""
        issuer = os.environ["CLERK_JWT_ISSUER_DOMAIN"].rstrip("/")
        try:
            header = jwt.get_unverified_header(token)
            public_key = await asyncio.to_thread(_fetch_public_key, header["kid"])
            claims = jwt.decode(
                token, public_key, algorithms=["RS256"], issuer=issuer, leeway=_JWT_LEEWAY_S,
                options={"verify_aud": False, "require": ["exp", "iat", "sub"]},
            )
        except jwt.ExpiredSignatureError: -> 401 "Token expired"
        except HTTPException: raise   # 401 unknown kid / 503 JWKS outage from _fetch_public_key
        except Exception as exc: log.warning(...); -> 401 "Invalid token"
        # azp AFTER the try (keep except-clause set identical: only ExpiredSignatureError,
        # HTTPException, Exception are referenced, so the test fakes need no extra attrs)
        allowed = dashboard_allowed_origins()
        azp = claims.get("azp")
        if allowed and azp is not None and str(azp).rstrip("/") not in allowed:
            log.warning("Clerk JWT azp %r not in DASHBOARD_ALLOWED_ORIGINS", azp)
            raise HTTPException(401, "Invalid token")
        return claims
    ```
