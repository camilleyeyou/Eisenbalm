```ts
/** Phase 51.1 — THE lean claim-row projection. Every dashboard surface that
 * slims claimChecks.listByRunId rows MUST use this (Pitfall 2: a projection
 * that drops `verdict` makes a contradicted row look legacy => client shows
 * nothing while the server blocks — the 9-day-outage drift class). */
export interface ClaimCheckDocLike {
  _id: string; _creationTime?: number; status: string; text: string
  sourceUrl?: string; sectionName?: string; claimIndex?: number; claimId?: string
  importance?: 'Load-bearing' | 'Supporting' | 'Incidental'
  changedSinceCheck?: boolean; conflict?: boolean; checkedAt?: number
  verdict?: string; verdictEvidence?: string; verdictSources?: string[]
}
export type DerivationClaimRow = NonNullable<DerivationInputs['claimRows']>[number]
export function projectClaimRow(row: ClaimCheckDocLike): DerivationClaimRow {
  return {
    _id: row._id, status: row.status, sourceUrl: row.sourceUrl, sectionName: row.sectionName,
    claimText: row.text, claimIndex: row.claimIndex, claimId: row.claimId, importance: row.importance,
    changedSinceCheck: row.changedSinceCheck, conflict: row.conflict, checkedAt: row.checkedAt,
    createdAt: row._creationTime, verdict: row.verdict, verdictEvidence: row.verdictEvidence,
    verdictSources: row.verdictSources,
  }
}
```
