        costUsd: v.number(), durationMs: v.number(), tokensIn: v.number(), tokensOut: v.number() }, ... })`
      → find row; if exists patch `{ status: 'done', completedAt, costUsd, durationMs, tokensIn, tokensOut }`

        completedAt: v.optional(v.number()) }, ... })`
      → find row; patch `{ status: 'failed', error, completedAt: completedAt ?? Date.now() }` else insert.

        outputSnapshot: v.optional(v.string()) }, ... })`
